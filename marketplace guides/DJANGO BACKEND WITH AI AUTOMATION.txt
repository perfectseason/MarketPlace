services.py

from openai import OpenAI
from django.conf import settings
from .prompts import SYSTEM_PROMPT

client = OpenAI(api_key=settings.OPENAI_API_KEY)


class ChatbotService:

    @staticmethod
    def generate(message, history=None):

        history = history or []

        messages = [
            {
                "role": "system",
                "content": SYSTEM_PROMPT
            }
        ]

        messages.extend(history)

        messages.append({
            "role": "user",
            "content": message
        })

        response = client.chat.completions.create(
            model="gpt-5.5",
            messages=messages,
            temperature=0.4,
        )

        return response.choices[0].message.content

exceptions.py


from rest_framework.views import exception_handler


def custom_exception_handler(exc, context):

    response = exception_handler(exc, context)

    if response is not None:
        response.data = {
            "success": False,
            "errors": response.data
        }

    return response

views.py


import logging

from openai import APIError

from rest_framework import status
from rest_framework.response import Response
from rest_framework.views import APIView

from .serializers import ChatSerializer
from .services import ChatbotService

logger = logging.getLogger(__name__)


class ChatbotAPIView(APIView):

    authentication_classes = []
    permission_classes = []

    def post(self, request):

        serializer = ChatSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        try:

            reply = ChatbotService.generate(
                serializer.validated_data["message"],
                serializer.validated_data.get("conversation")
            )

            return Response(
                {
                    "success": True,
                    "reply": reply
                },
                status=status.HTTP_200_OK
            )

        except APIError as e:

            logger.exception(e)

            return Response(
                {
                    "success": False,
                    "message": "AI service unavailable."
                },
                status=status.HTTP_503_SERVICE_UNAVAILABLE
            )

        except Exception as e:

            logger.exception(e)

            return Response(
                {
                    "success": False,
                    "message": "Internal server error."
                },
                status=status.HTTP_500_INTERNAL_SERVER_ERROR
            )

Production improvements


For a production deployment, I recommend adding:

> JWT authentication (if chats are tied to logged-in users).

> Per-user and per-IP rate limiting (e.g., django-ratelimit or DRF throttling).

> Structured logging with request IDs.

> Conversation persistence in PostgreSQL (models for conversations and messages).

> Streaming responses using Server-Sent Events (SSE) or WebSockets for a more responsive UI.

> Asynchronous processing with Celery for long-running tasks.

> Retry logic with exponential backoff for transient AI API failures.

> Input sanitization and prompt-injection mitigation.

> Token counting and conversation trimming to stay within model context limits.

> Usage analytics and monitoring (e.g., Sentry, Prometheus, OpenTelemetry).

> Environment-specific configuration and secret management.

> Comprehensive automated tests (unit, integration, and API tests).

> CORS configuration and CSRF protection appropriate to your frontend architecture.

- With those additions, the backend will be robust enough to support a production React/Vite frontend while remaining maintainable and scalable.

Example:


from openai import OpenAI
from django.conf import settings

client = OpenAI(api_key=settings.OPENAI_API_KEY)


class ChatbotService:

    @staticmethod
    def generate(prompt):

        response = client.responses.create(
            model="gpt-5o-mini",
            input=prompt,
            temperature=0.2,
            max_output_tokens=100,
        )

        return response.output_text

from rest_framework.views import APIView
from rest_framework.response import Response

from .serializers import ChatSerializer
from .services import ChatbotService


class ChatbotAPIView(APIView):

    def post(self, request):

        serializer = ChatSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        reply = ChatbotService.generate(
            serializer.validated_data["message"]
        )

        return Response({
            "success": True,
            "message": reply
        })

Your AI logic belongs here.



class ChatbotService:

    def generate(conversation, prompt):

        history = []

        for message in conversation.messages.all():

            history.append({
                "role": message.role,
                "content": message.content
            })

        history.append({
            "role": "user",
            "content": prompt
        })

        response = client.responses.create(
            model="gpt-5.5",
            input=history,
            temperature=0.2
        )

        return response.output_text

One recommendation before you start

Given your experience building full-stack applications with Django and React, I recommend building this in stages rather than trying to implement every feature at once:



Phase 1: Authentication (JWT) + Chat API + Database models.

Phase 2: Conversation history, rate limiting, logging, and prompt management.

Phase 3: Appointment scheduling integrated with your Django models and AI function-calling.

Phase 4: Streaming responses, voice input/output, notifications, and deployment.


This approach gives you a maintainable codebase and lets you test each layer before adding more complexity.
